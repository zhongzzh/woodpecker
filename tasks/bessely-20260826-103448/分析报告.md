### 文档自身缺陷审计

- 提示部分最后一句“$J_v(z)$是 bessely”应为“$J_v(z)$是 besselj”。
- scale 说明中“按因子 $exp((-abs(z)))$ 缩放”与语法部分“按因子 $exp(-abs(imag(z)))$”不一致，应为后者。
- 示例 `bessely.(3,0:5)` 中 nu=3 是 Int，文档 nu 数据类型列表未明确列出 Int（默认 Int 对应 Int64，可接受但建议明确）。
- z 数据类型列表包含 UInt8 至 UInt128，而 nu 仅写 UInt，存在一致性差异，但不影响判定。

### 缺失测试代码补全

```julia
@testset "bessely 补充" begin
    # 复用已验证的标量参考值，避免编造真值
    ref_nu1_z1 = -0.78121282130028891
    ref_nu2_z1 = -1.6506826068162548

    # nu 缺失数据类型：Int16、Int128、Float16、UInt
    @test isapprox(bessely(Int16(2), 1.0), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(Int128(2), 1.0), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test bessely(Float16(2), 1.0) isa Float64
    @test isapprox(bessely(Float16(2), 1.0), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test bessely(UInt(2), 1.0) isa Float64
    @test isapprox(bessely(UInt(2), 1.0), ref_nu2_z1; rtol=1e-13, atol=1e-13)

    # nu 矩阵结构
    nu_mat = [1 2; 1 2]
    expected_nu_mat = [
        ref_nu1_z1 ref_nu2_z1
        ref_nu1_z1 ref_nu2_z1
    ]
    y_nu_mat = bessely(nu_mat, 1.0)
    @test size(y_nu_mat) == size(nu_mat)
    @test isapprox(y_nu_mat, expected_nu_mat; rtol=1e-12, atol=1e-12)

    # nu N 维数组结构（三维）
    nu_3d = reshape([1, 2, 1, 2], 2, 1, 2)
    expected_nu_3d = reshape([
        ref_nu1_z1, ref_nu2_z1,
        ref_nu1_z1, ref_nu2_z1
    ], 2, 1, 2)
    y_nu_3d = bessely(nu_3d, 1.0)
    @test size(y_nu_3d) == size(nu_3d)
    @test isapprox(y_nu_3d, expected_nu_3d; rtol=1e-12, atol=1e-12)

    # z 缺失整数类型和无符号类型
    @test isapprox(bessely(2, 1), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, Int32(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, Int16(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, Int128(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, UInt8(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, UInt16(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, UInt32(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, UInt64(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)
    @test isapprox(bessely(2, UInt128(1)), ref_nu2_z1; rtol=1e-13, atol=1e-13)

    # z N 维数组结构（三维）
    z_3d = fill(1.0, 2, 2, 1)
    y_z_3d = bessely(2, z_3d)
    @test size(y_z_3d) == size(z_3d)
    @test isapprox(y_z_3d, fill(ref_nu2_z1, 2, 2, 1); rtol=1e-13, atol=1e-13)
end
```

静态检验说明：该代码块为一个独立的顶层 `@testset`，作用域自足，所有参考值和测试数据均在块内重新定义，不引用原测试块内部变量；仅使用文件顶层的 `using Test` 和 `using TyMathCore`（`bessely`、`isapprox` 可用）。代码块可整体追加到 `bessely.jl` 文件末尾，不需要插入或修改任何已有代码。
