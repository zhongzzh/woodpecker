    @test length(B) >= n
    for k in 1:length(B)
        @test size(B[k], 2) == 2
        @test eltype(B[k]) == Int64
    end

    if length(B_xy) > 0 && length(B_xy[1]) > 0
        # yx模式下第一列应为行坐标，xy模式下第一列应为列坐标，两者值应可区分

    @test size(L) == size(BW)
    @test eltype(L) == Float64
    @test n2 == length(B_noholes)
    @test maximum(L) == n2
