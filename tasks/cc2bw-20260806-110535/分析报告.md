中优先级

- CC 的 `Connectivity` 和 `PixelIdxList` 没有字段级结构断言。建议至少检查四个字段均存在、`length(CC.PixelIdxList) == CC.NumObjects`，以及 `CC.ImageSize == size(源图像)`。
- 正整数输入仅覆盖普通有效索引，没有验证零、负数、超出 `CC.NumObjects` 的索引。文档应先明确异常契约，再补充拒绝性测试。
- 文档没有明确具体整数类型。现有证据主要对应平台默认 `Int`；若准备支持无符号整数或其他位宽，应先更新文档，再分别补测。
- `[1 2]` 和 `[true false]` 是二维矩阵，属于文档外输入。应决定矩阵是否为正式支持形式；若不支持，应收紧实现签名和测试，若支持则应补充文档。

低优先级

- 正整数向量没有覆盖重复索引、乱序索引和空向量。实现会调用 `unique`，但文档未说明相应语义，需先确认公开契约。
- 没有明确测试零对象 CC。空连通分量属于边界场景，文档也未说明默认调用和空选择的返回行为。
- 没有验证输出与原始输入数组相互独立。文档未声明别名或复制语义，因此优先级较低。

@testset "cc2bw 补充覆盖" begin
    source_bw = falses(7, 7)
    source_bw[1, 1] = true
    source_bw[4, 4] = true
    source_bw[7, 7] = true

    cc = bwconncomp(source_bw)

    @testset "CC 四字段结构与默认转换" begin
        @test hasproperty(cc, :Connectivity)
        @test hasproperty(cc, :ImageSize)
        @test hasproperty(cc, :NumObjects)
        @test hasproperty(cc, :PixelIdxList)
        @test cc.ImageSize == size(source_bw)
        @test cc.NumObjects == 3
        @test length(cc.PixelIdxList) == cc.NumObjects

        actual = cc2bw(cc)
        @test eltype(actual) == Bool
        @test size(actual) == cc.ImageSize
        @test actual == source_bw
    end

    @testset "正整数标量与向量的选择语义" begin
        scalar_actual = cc2bw(cc; ObjectsToKeep=1)
        scalar_expected = falses(cc.ImageSize)
        scalar_expected[cc.PixelIdxList[1]] .= true
        @test scalar_actual == scalar_expected

        integer_selection = [1, 3]
        vector_actual = cc2bw(cc; ObjectsToKeep=integer_selection)
        vector_expected = falses(cc.ImageSize)
        for object_index in integer_selection
            vector_expected[cc.PixelIdxList[object_index]] .= true
        end
        @test vector_actual == vector_expected
    end

    @testset "等长逻辑向量的选择语义" begin
        logical_selection = Bool[true, false, true]
        @test logical_selection isa Vector{Bool}
        @test length(logical_selection) == cc.NumObjects

        logical_actual = cc2bw(cc; ObjectsToKeep=logical_selection)
        logical_expected = falses(cc.ImageSize)
        for object_index in findall(logical_selection)
            logical_expected[cc.PixelIdxList[object_index]] .= true
        end

        @test eltype(logical_actual) == Bool
        @test size(logical_actual) == cc.ImageSize
        @test logical_actual == logical_expected
        @test all(logical_actual[cc.PixelIdxList[1]])
        @test all(.!logical_actual[cc.PixelIdxList[2]])
        @test all(logical_actual[cc.PixelIdxList[3]])
    end

    @testset "逻辑向量长度契约" begin
        short_selection = Bool[true, false]
        @test length(short_selection) != cc.NumObjects

        short_selection_rejected = try
            cc2bw(cc; ObjectsToKeep=short_selection)
            false
        catch
            true
        end

        @test short_selection_rejected
    end
end
