chromadapt	using TyImageProcessing
using TyPlot
using BenchmarkTools

A = imread("hallway.jpg");
imshow(A)
title("Original Image")

x = 2800;
y = 1000;
gray_val = impixel(A, x, y);
hold("on")
plot(x, y, "."; color="g");
hold("off")

first_run_time = @elapsed begin
   B = chromadapt(A, gray_val);
end

second_run_time = @belapsed begin
   B = chromadapt(A, gray_val);
end

figure()
imshow(B)
title("White-Balanced Image")

println("首次用时： ",first_run_time)
println("二次用时： ",second_run_time)
