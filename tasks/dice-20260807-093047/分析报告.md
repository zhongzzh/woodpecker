@testset "dice 补充覆盖" begin
    # --- 二值图像模式补充 ---
    @testset "二值图像标量输出强断言" begin
        BW1 = rand(Bool, 50, 50)
        BW2 = rand(Bool, 50, 50)
        s = dice(BW1, BW2)
        @test s isa Float64
        @test s isa Real
        @test 0.0 <= s <= 1.0
    end

    @testset "二值图像一维输入" begin
        BW1 = rand(Bool, 100)
        BW2 = rand(Bool, 100)
        s = dice(BW1, BW2)
        @test s isa Float64
        @test 0.0 <= s <= 1.0
    end

    @testset "二值图像三维输入" begin
        BW1 = rand(Bool, 10, 10, 10)
        BW2 = rand(Bool, 10, 10, 10)
        s = dice(BW1, BW2)
        @test s isa Float64
        @test 0.0 <= s <= 1.0
    end

    @testset "二值图像大小不匹配" begin
        BW1 = rand(Bool, 50, 50)
        BW2 = rand(Bool, 30, 30)
        @test_throws DimensionMismatch dice(BW1, BW2)
    end

    @testset "二值图像全真与全假" begin
        BW_true = trues(20, 20)
        BW_false = falses(20, 20)
        # 全真 vs 全真，完全匹配
        @test dice(BW_true, BW_true) ≈ 1.0
        # 全真 vs 全假，无交集
        @test dice(BW_true, BW_false) ≈ 0.0
        # 全假 vs 全假，定义上通常为 0 或 NaN，此处只验证不抛出异常且返回标量
        s = dice(BW_false, BW_false)
        @test s isa Float64
    end

    # --- 标签图像模式补充 ---
    @testset "标签图像输出结构验证" begin
        # 零标签场景：全背景
        L1 = zeros(Int64, 50, 50)
        L2 = zeros(Int64, 50, 50)
        s_zero = dice(L1, L2)
        @test s_zero isa AbstractVecOrMat{Float64}
        # 零标签时通常返回空向量，或长度为0
        @test length(s_zero) == 0

        # 单标签场景
        L1 = ones(Int64, 50, 50)
        L2 = ones(Int64, 50, 50)
        s_single = dice(L1, L2)
        @test length(s_single) == 1
        @test s_single[1] ≈ 1.0
        @test all(0.0 .<= s_single .<= 1.0)

        # 多标签场景，验证输出长度等于非零标签数量
        L1 = Int64.(repeat(1:4, inner=(25, 25))[1:50, 1:50])
        L2 = deepcopy(L1)
        s_multi = dice(L1, L2)
        @test length(s_multi) == 4
        @test all(0.0 .<= s_multi .<= 1.0)
    end

    @testset "标签图像一维输入" begin
        L1 = Int64.([0, 1, 1, 2, 2, 0])
        L2 = Int64.([0, 1, 1, 2, 0, 0])
        s = dice(L1, L2)
        @test s isa AbstractVecOrMat{Float64}
        @test length(s) == 2
        @test all(0.0 .<= s .<= 1.0)
    end

    @testset "标签图像三维输入" begin
        L1 = Int64.(rand(1:3, 10, 10, 5))
        L2 = deepcopy(L1)
        s = dice(L1, L2)
        @test s isa AbstractVecOrMat{Float64}
        @test length(s) == 3
        @test all(0.0 .<= s .<= 1.0)
    end

    @testset "标签图像大小不匹配" begin
        L1 = Int64.(rand(1:3, 50, 50))
        L2 = Int64.(rand(1:3, 30, 30))
        @test_throws DimensionMismatch dice(L1, L2)
    end

    @testset "标签图像非连续标签值" begin
        L1 = Int64.([0, 2, 2, 10, 10, 0])
        L2 = Int64.([0, 2, 0, 10, 10, 0])
        s = dice(L1, L2)
        @test length(s) == 2
        @test all(0.0 .<= s .<= 1.0)
    end
end
