@testset "illumgray 补充2" begin
    A64 = cat(fill(0.1, 4, 4), fill(0.2, 4, 4), fill(0.4, 4, 4); dims=3)

    @test isapprox(illumgray(A64, Float32(0)), [0.1 0.2 0.4]; atol=1e-6)
    @test isapprox(illumgray(A64, UInt16(0)), [0.1 0.2 0.4]; atol=1e-12)

    @test isapprox(illumgray(A64, Float32[0, 0]), [0.1 0.2 0.4]; atol=1e-6)
    @test isapprox(illumgray(A64, UInt8[0, 0]), [0.1 0.2 0.4]; atol=1e-12)
    @test isapprox(illumgray(A64, UInt16[0, 0]), [0.1 0.2 0.4]; atol=1e-12)

    @test isapprox(illumgray(A64), illumgray(A64, 1); atol=1e-12)

    mask_f32 = ones(Float32, 4, 4)
    mask_f32[1:2, 1:2] .= 0
    @test isapprox(
        illumgray(A64; Mask=mask_f32, Norm=2), [0.1 0.2 0.4] ./ sqrt(12); atol=1e-6
    )

    mask_u16 = ones(UInt16, 4, 4)
    mask_u16[1:2, 1:2] .= 0
    @test isapprox(illumgray(A64; Mask=mask_u16), [0.1 0.2 0.4]; atol=1e-12)

    mask = trues(4, 4)
    mask[1:2, 1:2] .= false
    @test isapprox(
        illumgray(A64; Mask=mask, Norm=Float32(2)),
        illumgray(A64; Mask=mask, Norm=2);
        atol=1e-12,
    )
    @test isapprox(
        illumgray(A64; Mask=mask, Norm=UInt16(2)),
        illumgray(A64; Mask=mask, Norm=2);
        atol=1e-12,
    )

    @test isapprox(illumgray(A64, 50), [0.1 0.2 0.4]; atol=1e-12)
    @test isapprox(illumgray(A64, [50, 50]), [0.1 0.2 0.4]; atol=1e-12)
end
