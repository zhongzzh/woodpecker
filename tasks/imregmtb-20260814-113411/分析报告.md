### 缺失测试代码补全

```julia
@testset "imregmtb 补充" begin
    base = UInt8.(reshape(0:255, 16, 16))
    gray_u8 = repeat(base, 4, 4)
    rgb_u8 = Array{UInt8}(undef, 64, 64, 3)
    rgb_u8[:, :, 1] = gray_u8
    rgb_u8[:, :, 2] = reverse(gray_u8, dims=1)
    rgb_u8[:, :, 3] = reverse(gray_u8, dims=2)
    gray_f64 = Float64.(gray_u8) ./ 255
    rgb_f64 = Float64.(rgb_u8) ./ 255

    @testset "显式 UInt8 灰度" begin
        R, shift = imregmtb(gray_u8, gray_u8)
        @test size(R) == (64, 64) && eltype(R) == UInt8
        @test shift == zeros(1, 2)
    end

    @testset "显式 Float64 灰度" begin
        R, shift = imregmtb(gray_f64, gray_f64)
        @test size(R) == (64, 64) && eltype(R) == Float64
        @test shift == zeros(1, 2)
    end

    @testset "显式 UInt8 RGB" begin
        R, shift = imregmtb(rgb_u8, rgb_u8)
        @test size(R) == (64, 64, 3) && eltype(R) == UInt8
        @test shift == zeros(1, 2)
    end

    @testset "显式 Float64 RGB" begin
        R, shift = imregmtb(rgb_f64, rgb_f64)
        @test size(R) == (64, 64, 3) && eltype(R) == Float64
        @test shift == zeros(1, 2)
    end

    @testset "输入尺寸不一致异常" begin
        err = try
            imregmtb(zeros(UInt8, 64, 64), zeros(UInt8, 65, 64))
            nothing
        catch e
            e
        end
        @test err !== nothing
        msg = sprint(showerror, err)
        @test occursin("size must be", msg) || occursin("大小", msg)
    end
end
```
