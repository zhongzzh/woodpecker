### 缺失测试代码补全

```julia
@testset "imregmtb 补充" begin
    @testset "非 UInt8 类型多图非零位移" begin
        base = imread("office_6.jpg")

        base_f32 = im2single(base)
        moved_f32, = imtranslate(base_f32, reshape([3.0, -2.0], 1, 2))
        R1_f32, R2_f32, shift_f32 = imregmtb(base_f32, moved_f32, base_f32)
        @test size(R1_f32) == size(base_f32) && eltype(R1_f32) == Float32
        @test size(R2_f32) == size(moved_f32) && eltype(R2_f32) == Float32
        @test size(shift_f32) == (2, 2)
        @test eltype(shift_f32) == Float64
        @test !all(iszero, shift_f32)

        base_f64 = Float64.(base) ./ 255.0
        moved_f64, = imtranslate(base_f64, reshape([3.0, -2.0], 1, 2))
        R1_f64, R2_f64, shift_f64 = imregmtb(base_f64, moved_f64, base_f64)
        @test size(R1_f64) == size(base_f64) && eltype(R1_f64) == Float64
        @test size(R2_f64) == size(moved_f64) && eltype(R2_f64) == Float64
        @test size(shift_f64) == (2, 2)
        @test eltype(shift_f64) == Float64
        @test !all(iszero, shift_f64)

        base_u16 = im2uint16(base)
        moved_u16, = imtranslate(base_u16, reshape([3.0, -2.0], 1, 2))
        R1_u16, R2_u16, shift_u16 = imregmtb(base_u16, moved_u16, base_u16)
        @test size(R1_u16) == size(base_u16) && eltype(R1_u16) == UInt16
        @test size(R2_u16) == size(moved_u16) && eltype(R2_u16) == UInt16
        @test size(shift_u16) == (2, 2)
        @test eltype(shift_u16) == Float64
        @test !all(iszero, shift_u16)
    end

    @testset "最小尺寸成功路径" begin
        rng = MT19937ar(2024)
        img64 = UInt8.(rand(rng, 0:255, 64, 64))

        R64, shift64 = imregmtb(img64, img64)
        @test size(R64) == size(img64) && eltype(R64) == UInt8
        @test shift64 == zeros(1, 2)
    end
end
```
