@testset "imresize 补充覆盖" begin
    # -- 补充 A 的其他数据类型（UInt16, Int16, Float32, Float64）--
    img_u16 = UInt16.(rand(0:65535, 20, 30))
    @test size(imresize(img_u16, 0.5)) == ceil.(Int, size(img_u16) .* 0.5)
    @test eltype(imresize(img_u16, 0.5)) == UInt16

    img_i16 = Int16.(rand(-32768:32767, 15, 25))
    @test size(imresize(img_i16, [10, 10])) == (10, 10)
    @test eltype(imresize(img_i16, [10, 10])) == Int16

    img_f32 = rand(Float32, 20, 20)
    @test size(imresize(img_f32, 0.8)) == ceil.(Int, size(img_f32) .* 0.8)
    @test eltype(imresize(img_f32, 0.8)) == Float32

    img_f64 = rand(Float64, 10, 10)
    @test size(imresize(img_f64, 1.5)) == ceil.(Int, size(img_f64) .* 1.5)
    @test eltype(imresize(img_f64, 1.5)) == Float64

    # -- 补充 [numrows, numcols] 的 Float64 向量形式 --
    img = rand(UInt8, 20, 30)
    @test size(imresize(img, [12.0, 20.0])) == (12, 20)

    # -- 补充 scale 放大场景（>1）--
    img = rand(UInt8, 10, 10)
    @test size(imresize(img, 2.0)) == (20, 20)

    # -- 补充 A 为 Bool 类型（二值图像，对应文档说明）--
    img_bool = rand(Bool, 16, 16)
    resized_bool = imresize(img_bool, 0.5)
    @test size(resized_bool) == (8, 8)
    @test eltype(resized_bool) == Bool

    # -- 补充非 UInt8 类型搭配显式 method --
    img_f32_m = rand(Float32, 12, 12)
    resized_m = imresize(img_f32_m, 0.6; method="bilinear")
    @test size(resized_m) == ceil.(Int, size(img_f32_m) .* 0.6)
    @test eltype(resized_m) == Float32
end
