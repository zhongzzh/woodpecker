@testset "imresize 补充覆盖" begin
    # A 参数数据类型补充
    @testset "A 数据类型" begin
        # UInt16 灰度图像
        img_uint16 = rand(UInt16, 20, 30)
        resized_uint16 = imresize(img_uint16, 0.5)
        @test eltype(resized_uint16) == UInt16
        @test size(resized_uint16) == ceil.(Int, size(img_uint16) .* 0.5)

        # Int16 灰度图像
        img_int16 = rand(Int16, 20, 30)
        resized_int16 = imresize(img_int16, 0.5)
        @test eltype(resized_int16) == Int16
        @test size(resized_int16) == ceil.(Int, size(img_int16) .* 0.5)

        # Float32 灰度图像
        img_float32 = rand(Float32, 20, 30)
        resized_float32 = imresize(img_float32, 0.5)
        @test eltype(resized_float32) == Float32
        @test size(resized_float32) == ceil.(Int, size(img_float32) .* 0.5)

        # Float64 灰度图像
        img_float64 = rand(Float64, 20, 30)
        resized_float64 = imresize(img_float64, 0.5)
        @test eltype(resized_float64) == Float64
        @test size(resized_float64) == ceil.(Int, size(img_float64) .* 0.5)

        # Bool 二值图像
        img_bool = rand(Bool, 20, 30)
        resized_bool = imresize(img_bool, 0.5)
        @test eltype(resized_bool) == Bool
        @test size(resized_bool) == ceil.(Int, size(img_bool) .* 0.5)

        # UInt16 彩色图像
        img_uint16_3d = rand(UInt16, 20, 30, 3)
        resized_uint16_3d = imresize(img_uint16_3d, 0.5)
        @test eltype(resized_uint16_3d) == UInt16
        @test size(resized_uint16_3d) == (ceil(Int, 20 * 0.5), ceil(Int, 30 * 0.5), 3)

        # Float32 彩色图像
        img_float32_3d = rand(Float32, 20, 30, 3)
        resized_float32_3d = imresize(img_float32_3d, 0.5)
        @test eltype(resized_float32_3d) == Float32
        @test size(resized_float32_3d) == (ceil(Int, 20 * 0.5), ceil(Int, 30 * 0.5), 3)
    end

    # scale 参数补充
    @testset "scale 参数" begin
        img = rand(UInt8, 20, 30)

        # 放大场景
        resized_up = imresize(img, 2.0)
        @test size(resized_up) == (40, 60)

        # 异常值 0
        @test_throws Exception imresize(img, 0.0)

        # 异常值负数
        @test_throws Exception imresize(img, -0.5)
    end

    # [numrows, numcols] 参数补充
    @testset "[numrows, numcols] 参数" begin
        img = rand(UInt8, 20, 30)

        # Float64 向量
        resized_float64_sz = imresize(img, [12.0, 20.0])
        @test size(resized_float64_sz) == (12, 20)

        # 含 NaN 向量，自动计算维度
        resized_nan1 = imresize(img, [NaN, 20.0])
        @test size(resized_nan1)[2] == 20
        @test size(resized_nan1)[1] > 0

        resized_nan2 = imresize(img, [12.0, NaN])
        @test size(resized_nan2)[1] == 12
        @test size(resized_nan2)[2] > 0

        # 异常输入：长度不为 2
        @test_throws Exception imresize(img, [12.0])
        @test_throws Exception imresize(img, [12.0, 20.0, 30.0])

        # 异常输入：负数
        @test_throws Exception imresize(img, [-12.0, 20.0])
    end

    # method 参数补充
    @testset "method 参数" begin
        img = rand(UInt8, 20, 30)

        # method 作为位置参数
        resized_positional = imresize(img, 0.5, "nearest")
        @test size(resized_positional) == ceil.(Int, size(img) .* 0.5)

        # 无效方法字符串
        @test_throws Exception imresize(img, 0.5; method="invalid_method")
    end

    # 输出类型一致性补充：非 UInt8 图像的 method 测试
    @testset "非 UInt8 图像的插值方法" begin
        img_float32 = rand(Float32, 20, 30)
        for m in ["nearest", "bilinear", "bicubic", "area", "lanczos"]
            resized = imresize(img_float32, 0.5; method=m)
            @test eltype(resized) == Float32
            @test size(resized) == ceil.(Int, size(img_float32) .* 0.5)
        end

        img_bool = rand(Bool, 20, 30)
        for m in ["nearest", "bilinear", "bicubic", "area", "lanczos"]
            resized = imresize(img_bool, 0.5; method=m)
            @test eltype(resized) == Bool
            @test size(resized) == ceil.(Int, size(img_bool) .* 0.5)
        end
    end
end
