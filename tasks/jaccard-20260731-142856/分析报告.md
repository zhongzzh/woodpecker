@testset "jaccard 补充覆盖" begin
    # --- 一维 Bool 向量 ---
    v1 = Bool[1, 0, 1, 0]
    v2 = Bool[0, 1, 1, 0]
    sim_vec = jaccard(v1, v2)
    @test sim_vec isa Float64

    # --- 三维 Bool 数组 ---
    b3d_1 = rand(Bool, 4, 4, 4)
    b3d_2 = rand(Bool, 4, 4, 4)
    sim_3d = jaccard(b3d_1, b3d_2)
    @test sim_3d isa Float64

    # --- 一维 Int64 标签 ---
    l1_vec = Int64[1, 0, 2, 2]
    l2_vec = Int64[1, 1, 2, 0]
    sim_lvec = jaccard(l1_vec, l2_vec)
    @test sim_lvec isa AbstractVecOrMat{Float64}
    @test size(sim_lvec, 1) == 2

    # --- 尺寸不匹配 (Bool) ---
    @test_throws ErrorException jaccard(falses(3, 3), falses(3, 4))

    # --- 尺寸不匹配 (Int64) ---
    @test_throws ErrorException jaccard(Int64.(zeros(3, 3)), Int64.(zeros(3, 4)))

    # --- 类型不匹配 ---
    @test_throws ErrorException jaccard(falses(3, 3), Int64.(zeros(3, 3)))

    # --- 空数组 (Bool) ---
    @test_throws ErrorException jaccard(Bool[], Bool[])

    # --- 空数组 (Int64) ---
    @test_throws ErrorException jaccard(Int64[], Int64[])

    # --- Float64 标签异常校验：NaN ---
    Lnan = Float64[1 NaN; 0 2]
    Lval = Float64[1 0; 2 2]
    @test_throws ErrorException jaccard(Lnan, Lval)
    @test_throws ErrorException jaccard(Lval, Lnan)

    # --- Float64 标签异常校验：Inf ---
    Linf = Float64[1 Inf; 0 2]
    @test_throws ErrorException jaccard(Linf, Lval)
    @test_throws ErrorException jaccard(Lval, Linf)

    # --- Float64 标签异常校验：负数 ---
    Lneg = Float64[1 -1; 0 2]
    @test_throws ErrorException jaccard(Lneg, Lval)
    @test_throws ErrorException jaccard(Lval, Lneg)

    # --- Float64 标签异常校验：非整数 ---
    Lfrac = Float64[1 1.5; 0 2]
    @test_throws ErrorException jaccard(Lfrac, Lval)
    @test_throws ErrorException jaccard(Lval, Lfrac)

    # --- Float64 标签缺失类别 NaN 行为 ---
    L_f64_a = Float64[1 3; 1 3]
    L_f64_b = Float64[1 3; 1 3]
    sim_f64_missing = jaccard(L_f64_a, L_f64_b)
    @test sim_f64_missing[1] == 1.0
    @test isnan(sim_f64_missing[2])
    @test sim_f64_missing[3] == 1.0
