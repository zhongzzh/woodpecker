@testset "jaccard 补充覆盖" begin
    # ===== 高优先级: 错误路径测试 =====
    @testset "错误路径: 输入为空" begin
        emptyBool = Bool[]
        @test_throws ErrorException jaccard(emptyBool, emptyBool)
        emptyInt = Int64[]
        @test_throws ErrorException jaccard(emptyInt, emptyInt)
    end

    @testset "错误路径: 大小不匹配" begin
        A = rand(Bool, 10, 10)
        B = rand(Bool, 10, 5)
        @test_throws ErrorException jaccard(A, B)
    end

    @testset "错误路径: 类型不匹配" begin
        A = rand(Bool, 10, 10)
        B = rand(Int64, 10, 10)
        @test_throws MethodError jaccard(A, B)
    end

    @testset "错误路径: 标签图像含负值" begin
        A = Int64[-1 0 1; 2 3 4]
        B = Int64[0 1 2; 2 3 4]
        @test_throws ErrorException jaccard(A, B)
    end

    @testset "错误路径: 标签图像含非整数Float64" begin
        A = Float64[0.5 1.0; 2.0 3.0]
        B = Float64[1.0 2.0; 2.0 3.0]
        @test_throws ErrorException jaccard(A, B)
    end

    @testset "错误路径: 标签图像含NaN" begin
        A = Float64[NaN 1.0; 2.0 3.0]
        B = Float64[1.0 2.0; 2.0 3.0]
        @test_throws ErrorException jaccard(A, B)
    end

    @testset "错误路径: 标签图像含Inf" begin
        A = Float64[Inf 1.0; 2.0 3.0]
        B = Float64[1.0 2.0; 2.0 3.0]
        @test_throws ErrorException jaccard(A, B)
    end

    # ===== 高优先级: BitMatrix 类型 =====
    @testset "BitMatrix 二值输入" begin
        # BitMatrix 与自身相同，应得 1.0
        BA = trues(4, 4)
        @test jaccard(BA, BA) ≈ 1.0

        # BitMatrix 全 false 与全 false
        BZ = falses(4, 4)
        @test isnan(jaccard(BZ, BZ))

        # BitMatrix 与 Matrix{Bool} 混合
        boolMat = rand(Bool, 5, 5)
        bitMat = BitMatrix(boolMat)
        @test jaccard(boolMat, bitMat) ≈ jaccard(bitMat, boolMat)
    end

    # ===== 中优先级: 高维数组 =====
    @testset "高维数组" begin
        # 1D Bool 向量
        BW1d_a = Bool[1, 0, 1, 1, 0]
        BW1d_b = Bool[1, 0, 0, 1, 0]
        sim_1d = jaccard(BW1d_a, BW1d_b)
        @test typeof(sim_1d) == Float64

        # 3D Bool 数组
        BW3d_a = rand(Bool, 8, 8, 3)
        BW3d_b = rand(Bool, 8, 8, 3)
        @test typeof(jaccard(BW3d_a, BW3d_b)) == Float64

        # 1D Int64 向量
        L1d_a = Int64[0, 1, 2, 1, 3]
        L1d_b = Int64[0, 2, 2, 1, 0]
        @test typeof(jaccard(L1d_a, L1d_b)) <: AbstractVecOrMat{Float64}
    end

    # ===== 中优先级: 全 true/全 false 边界 =====
    @testset "二值极端情况" begin
        # 全 true vs 全 true -> 1.0
        allTrue = trues(10, 10)
        @test jaccard(allTrue, allTrue) ≈ 1.0

        # 全 false vs 全 false -> NaN
        allFalse = falses(10, 10)
        @test isnan(jaccard(allFalse, allFalse))

        # 全 true vs 全 false -> 0.0
        @test jaccard(allTrue, allFalse) ≈ 0.0

        # 单元素 true 矩阵
        single1 = Bool[1]
        single0 = Bool[0]
        @test jaccard(single1, single1) ≈ 1.0
        @test jaccard(single0, single0) |> isnan
        @test jaccard(single1, single0) ≈ 0.0
    end

    # ===== 类型兼容性: Float64 标签（文档未声明但实现支持）=====
    @testset "Float64 标签常规输入" begin
        L1 = Float64[1.0 2.0; 3.0 1.0]
        L2 = Float64[1.0 1.0; 3.0 2.0]
        result = jaccard(L1, L2)
        # 手动计算: 标签1 交集2并集3→2/3≈0.6667; 标签2 交集0并集2→0.0; 标签3 交集1并集2→0.5
        # 注意标签顺序: 1,2,3
        expected = Float64[2/3; 0.0; 0.5]
        @test result ≈ expected
    end

    # ===== 验证 Dice 关系 =====
    @testset "与Dice指数关系" begin
        # jaccard = dice / (2 - dice)
        dice_val(A, B) = begin
            inter = sum(A .& B)
            total = sum(A) + sum(B)
            2 * inter / total
        end

        A = rand(Bool, 20, 20)
        B = rand(Bool, 20, 20)
        j = jaccard(A, B)
        d = dice_val(A, B)
        if !isnan(d) && d != 2.0
            @test j ≈ d / (2 - d)
        end
    end
end
