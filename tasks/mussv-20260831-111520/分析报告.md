### 缺失测试代码补全

```julia
@testset "mussv 补充" begin
    # FRD 模型输入
    frd_data = reshape(ComplexF64[0.5 + 0.1im, 0.6 - 0.2im], 1, 1, 2)
    frd_obj = frd(frd_data, [1.0, 2.0], 0.0)
    bounds_frd, _ = mussv(frd_obj, [-1 0])
    @test bounds_frd isa FRD
    @test size(bounds_frd.ResponseData) == (1, 2, 2)
    @test all(isfinite, bounds_frd.ResponseData)
    @test 0 <= bounds_frd.ResponseData[1, 2, 1] <= bounds_frd.ResponseData[1, 1, 1]

    # Options 选项补测
    Mopt = ComplexF64[0.3+0.1im 0.2-0.05im; -0.1+0.2im 0.4-0.3im]
    blocks_opt = [-1 0; 1 1]
    for opt in ("f", "U", "i", "s", "d", "x")
        b, _ = mussv(Mopt, blocks_opt, opt)
        @test size(b) == (1, 2)
        @test all(isfinite, b)
        @test 0 <= b[2] <= b[1]
    end
    b_m3, _ = mussv(Mopt, blocks_opt, "m3")
    @test size(b_m3) == (1, 2)
    @test all(isfinite, b_m3)
    @test 0 <= b_m3[2] <= b_m3[1]
    b_combo, _ = mussv(Mopt, blocks_opt, "fp")
    @test size(b_combo) == (1, 2)
    @test all(isfinite, b_combo)
    @test 0 <= b_combo[2] <= b_combo[1]

    # 重复实标量 r>1
    Mrep = Float64[0.1 0.2; 0.3 0.4]
    b_rep, _ = mussv(Mrep, [-2 0])
    @test size(b_rep) == (1, 2)
    @test all(isfinite, b_rep)
    @test 0 <= b_rep[2] <= b_rep[1]
end
```
