### 覆盖缺口与优先级

**高优先级**
- ConstrainedTriangulation 作为 TR 输入：文档明确列为允许类型，但零测试覆盖。
- c 与 z 形状/类型不一致但元素数量相同：文档仅要求“元素数量相同”，现有测试全为同形同类，未验证合法性。
- 返回值 h 的类型断言与属性可读性：文档承诺返回补片对象可查询/修改，现有测试仅弱检查 `!== nothing`。
- T 的非整数索引、空矩阵、单行矩阵：边界与异常路径完全缺失。

**中优先级**
- facecolor/edgecolor 的 "flat"、"none"、默认值显式传入：文档列为允许值，测试未覆盖。
- linestyle="none"：文档枚举值，测试未覆盖。
- linewidth 默认值 0.5 显式传入与非正值异常：文档约束“正值”，测试未覆盖边界。
- x/y/z/c 的非 Float64 类型 (Int、Float32 等)：文档未限制类型，测试仅覆盖 Float64。
- facealpha 越界值 (<0、>1)：文档约束 [0,1]，测试未验证错误处理。

**低优先级**
- x/y/z/c 空向量/矩阵：极端边界。
- x/y/z 含 NaN/Inf：特殊数值行为。
- 组合 name-value 参数的更多排列：test8 已覆盖一种组合，其余排列优先级较低。

### 缺失测试代码补全

```julia
@testset "trisurf 补充覆盖" begin
    using TyComputationalGeometry
    using TyComputationalGeometryPlot
    using TyPlot
    using TyPlotTest
    using TyBase
    using Test

    # 构造基础网格数据，复用现有风格
    x, y = meshgrid2(1:10, 1:10)
    _, _, z = peaks(10)
    T = delaunay(x, y)

    # ---- 1. ConstrainedTriangulation 作为 TR 输入 ----
    @testset "ConstrainedTriangulation 输入" begin
        # 构造约束边：取前两个点为约束
        P = hcat(x[:], y[:], z[:])
        constraints = [1 2]  # 简单约束示例
        CT = ConstrainedTriangulation(T, P, constraints)
        h = trisurf(CT)
        @test h !== nothing
        @test h isa TyPlot.Patch  # 验证返回类型
    end

    # ---- 2. c 与 z 形状/类型不一致但元素数量相同 ----
    @testset "c 与 z 异形同量" begin
        # z 为矩阵，c 为向量
        c_vec = vec(z)
        @test length(c_vec) == length(z)
        h1 = trisurf(T, x, y, z, c_vec)
        @test h1 !== nothing

        # z 为向量，c 为矩阵 (reshape)
        xv, yv, zv = x[:], y[:], z[:]
        Tv = delaunay(xv, yv)
        c_mat = reshape(zv, 2, length(zv)÷2)
        @test length(c_mat) == length(zv)
        h2 = trisurf(Tv, xv, yv, zv, c_mat)
        @test h2 !== nothing

        # c 为 Float32，z 为 Float64
        c_f32 = Float32.(vec(z))
        h3 = trisurf(T, x, y, z, c_f32)
        @test h3 !== nothing
    end

    # ---- 3. 返回值 h 类型与属性可读性 ----
    @testset "返回值 Patch 对象类型与属性" begin
        h = trisurf(T, x, y, z)
        @test h isa TyPlot.Patch
        # 读取关键属性验证可查询
        @test hasproperty(h, :facecolor)
        @test hasproperty(h, :facealpha)
        @test hasproperty(h, :edgecolor)
        @test hasproperty(h, :linewidth)
        @test hasproperty(h, :linestyle)
        # 修改属性验证可写
        h.facecolor = "red"
        h.facealpha = 0.3
        @test h.facecolor == "red"
        @test h.facealpha == 0.3
    end

    # ---- 4. T 边界结构：显式单行、空矩阵 (预期抛错或空图) ----
    @testset "T 边界结构" begin
        # 单行三角形
        T1 = reshape([1,2,3], 1, 3)
        x1 = [0.0, 1.0, 0.0]
        y1 = [0.0, 0.0, 1.0]
        z1 = [0.0, 0.0, 0.0]
        h1 = trisurf(T1, x1, y1, z1)
        @test h1 !== nothing

        # 空三角形矩阵：预期不报错但生成空补片
        T0 = Matrix{Int}(undef, 0, 3)
        x0 = Float64[]
        y0 = Float64[]
        z0 = Float64[]
        h0 = trisurf(T0, x0, y0, z0)
        @test h0 !== nothing
    end

    # ---- 5. x/y/z/c 非 Float64 类型 ----
    @testset "非 Float64 坐标与颜色" begin
        xi = Int.(x)
        yi = Int.(y)
        zi = Int.(z)
        Ti = delaunay(xi, yi)
        h1 = trisurf(Ti, xi, yi, zi)
        @test h1 !== nothing

        xf32 = Float32.(x)
        yf32 = Float32.(y)
        zf32 = Float32.(z)
        Tf32 = delaunay(xf32, yf32)
        h2 = trisurf(Tf32, xf32, yf32, zf32)
        @test h2 !== nothing
    end

    # ---- 6. name-value 缺失枚举值显式测试 ----
    @testset "name-value 缺失枚举" begin
        # facecolor: "flat" 显式、"none"
        h1 = trisurf(T, x, y, z; facecolor="flat")
        @test h1 !== nothing
        h2 = trisurf(T, x, y, z; facecolor="none")
        @test h2 !== nothing

        # edgecolor: "none"、默认 [0,0,0] 显式
        h3 = trisurf(T, x, y, z; edgecolor="none")
        @test h3 !== nothing
        h4 = trisurf(T, x, y, z; edgecolor=[0.0,0.0,0.0])
        @test h4 !== nothing

        # linestyle: "none"
        h5 = trisurf(T, x, y, z; linestyle="none")
        @test h5 !== nothing

        # linewidth: 默认 0.5 显式
        h6 = trisurf(T, x, y, z; linewidth=0.5)
        @test h6 !== nothing
    end

    # ---- 7. facealpha 越界值预期报错 ----
    @testset "facealpha 越界" begin
        @test_throws Exception trisurf(T, x, y, z; facealpha=-0.1)
        @test_throws Exception trisurf(T, x, y, z; facealpha=1.1)
    end

    # ---- 8. linewidth 非正值预期报错 ----
    @testset "linewidth 非正" begin
        @test_throws Exception trisurf(T, x, y, z; linewidth=0.0)
        @test_throws Exception trisurf(T, x, y, z; linewidth=-1.0)
    end
end
```

**静态检验说明**：上方代码块为单一顶层 `@testset "trisurf 补充覆盖"`，作用域自足，仅复用文件顶层已导入的模块与 `meshgrid2`、`peaks`、`delaunay`、`triangulation`、`ConstrainedTriangulation` 等公开函数；未引用原测试块内部局部变量；可整体追加到 `trisurf.jl` 物理末尾，无需任何插入或改写。

**仍需 baseline 校准项**：
