### 缺失测试代码补全

```julia
@testset "wcgain 补充" begin
    # 重建基础不确定系统，避免引用原测试块内局部变量
    P_supp = uss(tf(1, [1, 0])) + ultidyn("delta_supp", [1, 1]; Bound=0.4)
    K_supp = tf(2, [1 / 50, 1.0])
    S_supp = feedback(uss(1), P_supp * K_supp)

    # focus 连续时间 wmax=Inf 与 wmin=0 的合法边界
    inf_gain, _, inf_info = wcgain(S_supp, (0.0, Inf))
    @test inf_gain["CriticalFrequency"] >= 0.0
    @test all(0.0 .<= inf_info["Frequency"])
    @test size(inf_info["Bounds"]) == (length(inf_info["Frequency"]), 2)
    @test all(inf_info["Bounds"][:, 1] .<= inf_info["Bounds"][:, 2])

    # w 去重并升序排列
    unsorted_freq = [10.0, 1.0, 10.0, 5.0]
    sorted_gain, _, sorted_info = wcgain(S_supp, unsorted_freq)
    @test sorted_info["Frequency"] == [1.0, 5.0, 10.0]
    @test size(sorted_info["Bounds"]) == (3, 2)
    @test all(sorted_info["Bounds"][:, 1] .<= sorted_info["Bounds"][:, 2])
    @test sorted_gain["CriticalFrequency"] in sorted_info["Frequency"]

    # 离散时间系统 Nyquist 边界
    discrete_sys = uss(ss(tf(1.0, [1.0, -0.5], 1.0)))
    nyquist_focus = (0.0, float(pi))
    discrete_gain, _, discrete_info = wcgain(discrete_sys, nyquist_focus)
    @test discrete_gain["CriticalFrequency"] >= 0.0
    @test all(0.0 .<= discrete_info["Frequency"] .<= float(pi))
    @test size(discrete_info["Bounds"]) == (length(discrete_info["Frequency"]), 2)
    @test_throws ErrorException wcgain(discrete_sys, (0.0, float(pi) + 0.1))
end
```
