@testset "writeVideo 补充" begin
    # --- 二维缺失类型 (Motion JPEG 2000) ---
    @testset "二维灰度：UInt8, Int8, Int16, UInt16" begin
        for T in (UInt8, Int8, Int16, UInt16)
            outfile = tempname() * ".mj2"
            try
                img = T.(rand(0:typemax(T)>>2, 100, 100))
                v = VideoWriter(outfile, [100, 100])

                writeVideo(v, img)
                close(v)
